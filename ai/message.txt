% ─── S1 · settings.json y scopes ─────────────────────────────────────────────
\begin{frame}{settings.json y scopes\rf{1}}

  {\small \quees el archivo de configuración \textbf{declarativa} de Claude Code. Se carga desde 4 lugares; si un campo aparece en varios, gana el \textbf{más específico}.}\\[3pt]
  {\small \paraque fijar la config una vez en lugar de pedirla en prosa cada sesión. Cada key top-level configura un subsistema — es el índice del resto de la charla.}\\[8pt]

  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \textbf{Los 4 scopes} \small(managed $>$ local $>$ project $>$ user)\normalsize\\[4pt]
      \scriptsize
      \begin{tabular}{ll}
        \texttt{managed} & \texttt{/Library/.../ClaudeCode/} (org, no override) \\
        \texttt{user}    & \texttt{\textasciitilde/.claude/settings.json} (todos tus proyectos) \\
        \texttt{project} & \texttt{.claude/settings.json} (commiteado, equipo) \\
        \texttt{local}   & \texttt{.claude/settings.local.json} (gitignored) \\
      \end{tabular}
    \end{column}
    \begin{column}{0.44\textwidth}
      \textbf{Los grupos}\\[4pt]
      \scriptsize \texttt{model}\dd\texttt{permissions}\dd\texttt{hooks}\dd\texttt{statusLine}\dd\texttt{editorMode}\dd\texttt{theme}\dd\texttt{env}\dd\texttt{enabledPlugins}.
    \end{column}
  \end{columns}
  \vspace{6pt}
  \footnotesize $\rightarrow$ \dotf{settings.json}{mi \texttt{settings.json} completo en dotfiles}
\end{frame}

\section{Instrucciones y contexto}

% ─── S2 · CLAUDE.md / Rules ──────────────────────────────────────────────────
\begin{frame}{CLAUDE.md / Rules\rf{2}}

  {\small \quees instrucciones en markdown que Claude lee al inicio de \textbf{cada} sesión. \textbf{Rules} son lo mismo, pero se cargan \textbf{solo cuando aplican} al archivo que Claude está tocando.}\\[3pt]
  {\small \paraque darle estándares, arquitectura y comandos del proyecto de forma persistente, sin repetirlos.}\\[6pt]

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Orden de carga (se concatenan)}\\[3pt]
      \small
      \begin{enumerate}
        \item \texttt{/etc/claude-code/CLAUDE.md} \hfill managed
        \item \texttt{\textasciitilde/.claude/CLAUDE.md} \hfill usuario
        \item \texttt{./CLAUDE.md} o \texttt{.claude/CLAUDE.md} \hfill proyecto
        \item \texttt{./CLAUDE.local.md} \hfill gitignored
      \end{enumerate}
      \small \texttt{@path/file} importa (max 4 hops).
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Rules con path-scoping}\\[3pt]
      \small Archivos en \texttt{.claude/rules/*.md}. Con frontmatter \texttt{paths:} se cargan solo cuando Claude toca un archivo que matchea el glob — lazy-load por lenguaje.\\[8pt]
      \alert{Regla práctica:} CLAUDE.md \textbf{$<$200 líneas}. Más largo = Claude lo ignora. Lo específico va a \texttt{rules/}.
    \end{column}
  \end{columns}
\end{frame}

% ─── S3 · Memory ─────────────────────────────────────────────────────────────
\begin{frame}{Memory\rf{3}}

  {\small \quees estado que Claude escribe y relee \textbf{entre sesiones}: lo que aprendió de tu repo — comandos, decisiones, preferencias.}\\[3pt]
  {\small \paraque que no se lo repitas cada vez. CLAUDE.md = instrucciones estables; memory = estado que evoluciona.}\\[6pt]

  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Cómo funciona}\\[3pt]
      \small
      \begin{itemize}
        \item Directorio por repo: \texttt{\textasciitilde/.claude/projects/<repo>/memory/}
        \item \texttt{MEMORY.md} — índice; sus primeras \textbf{200 líneas / 25KB} se cargan cada sesión
        \item Topic files — Claude los carga on-demand según relevancia
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \textbf{Tipos recomendados}\\[3pt]
      \small
      \begin{tabular}{ll}
        \texttt{user}      & perfil y preferencias \\
        \texttt{feedback}  & qué corregir la próxima \\
        \texttt{project}   & estado, decisiones, fechas \\
        \texttt{reference} & links a sistemas externos \\
      \end{tabular}
      \vspace{6pt}
      \footnotesize En sesión: \texttt{/memory} navega, \texttt{/context} muestra qué se cargó.
    \end{column}
  \end{columns}
\end{frame}

% ─── S4 · Permissions ────────────────────────────────────────────────────────
\begin{frame}{Permissions\rf{4}}

  {\small \quees reglas declarativas de qué tools se \textbf{auto-aprueban}, cuáles \textbf{preguntan} y cuáles se \textbf{prohíben}. Es el filtro declarativo — el hermano de hooks.}\\[3pt]
  {\small \paraque controlar qué puede hacer Claude sin aprobar clic por clic.}\\[6pt]

  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Sintaxis de reglas}\\[3pt]
      \small
      \begin{itemize}
        \item \texttt{Bash(npm run build)} — match exacto
        \item \texttt{Bash(git *)} — prefijo, cualquier subcomando
        \item \texttt{WebFetch(domain:github.com)} — por dominio
        \item \texttt{mcp\_\_jira\_\_*} — todos los tools de un MCP
      \end{itemize}
      \vspace{4pt}
      \textbf{Evaluación:} deny $>$ ask $>$ allow \small(first match wins).
    \end{column}
    \begin{column}{0.44\textwidth}
      \textbf{Modos de permiso}\\[3pt]
      \small
      \begin{tabular}{ll}
        \texttt{default}    & prompt en primer uso \\
        \texttt{acceptEdits}& auto-acepta edits \\
        \texttt{plan}       & read-only \\
        \texttt{bypassPermissions} & sin prompts \\
      \end{tabular}
      \vspace{6pt}
      \footnotesize \textbf{Deny sin especificar} saca el tool del contexto de Claude por completo.
    \end{column}
  \end{columns}
\end{frame}

% ─── S5 · Hooks — eventos ────────────────────────────────────────────────────
\begin{frame}{Hooks — eventos\rf{5}}

  {\small \quees scripts que Claude Code dispara \textbf{automáticamente} en puntos fijos del ciclo de vida. A diferencia de todo lo demás, no es una sugerencia al modelo: \textbf{corre sí o sí, Claude no puede saltárselo}.}\\[3pt]
  {\small \paraque elegir el momento exacto donde intervenir. Es la única palanca \textbf{determinística} para condicionar el comportamiento de un agente.}\\[6pt]

  \small
  \begin{tabular}{lll}
    \toprule
    \textbf{Evento} & \textbf{Cuándo} & \textbf{Matcher} \\
    \midrule
    \texttt{SessionStart}      & inicio / resume / compact / clear & \texttt{startup}, \texttt{resume}… \\
    \texttt{UserPromptSubmit}  & antes de procesar el prompt & — \\
    \texttt{PreToolUse}        & antes de ejecutar un tool \textbf{(acá bloqueás)} & nombre del tool \\
    \texttt{PostToolUse}       & después de tool exitoso & nombre del tool \\
    \texttt{Stop}              & Claude termina su turno & — \\
    \texttt{PreCompact}        & antes de compactar el contexto & — \\
    \texttt{SubagentStart/Stop}& subagente spawneado / terminado & tipo de agente \\
    \texttt{SessionEnd}        & sesión termina & \texttt{clear}, \texttt{logout}… \\
    \bottomrule
  \end{tabular}
\end{frame}

% ─── S6 · Hooks — tipos y matcher ────────────────────────────────────────────
\begin{frame}{Hooks — tipos y matcher\rf{5}}

  {\small Cada hook tiene un \textbf{tipo} (qué ejecuta) y un \textbf{matcher} (qué lo dispara).}\\[6pt]

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{5 tipos de hook}\\[3pt]
      \small
      \begin{tabular}{ll}
        \toprule
        \textbf{Tipo} & \textbf{Qué hace} \\
        \midrule
        \texttt{command} & script bash; exit 0=ok, 2=bloquear \\[2pt]
        \texttt{http}    & POST JSON a un endpoint \\[2pt]
        \texttt{mcp\_tool} & llama tool de un MCP server \\[2pt]
        \texttt{prompt}  & Claude decide sí/no (timeout 30s) \\[2pt]
        \texttt{agent}   & subagente verifica (timeout 60s) \\
        \bottomrule
      \end{tabular}
      \vspace{6pt}
      \small\alert{El mecanismo determinístico:} exit \texttt{2} $\rightarrow$ la acción se \textbf{bloquea}, Claude ve el motivo por stderr y reintenta distinto.
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Matcher} — qué selecciona el hook\\[4pt]
      \small Alfanumérico = exact match. Cualquier otro carácter = regex JS.\\[4pt]
      \begin{tabular}{ll}
        \texttt{"Write"}           & tool exacto \\
        \texttt{"Edit|Write"}      & regex OR \\
        \texttt{"mcp\_\_jira\_\_*"}& todos los tools de Jira \\
        \texttt{"startup"}         & SessionStart al inicio \\
      \end{tabular}
      \vspace{6pt}
      \footnotesize \textbf{Criterio:} \texttt{command} para lógica local; \texttt{prompt}/\texttt{agent} para juicio; \texttt{http}/\texttt{mcp\_tool} para sistemas externos.
    \end{column}
  \end{columns}
\end{frame}

% ─── S7 · Hooks — ejemplo ────────────────────────────────────────────────────
\begin{frame}{Hooks — ejemplo\rf{5}}

  {\small Dos hooks reales, cada uno enforza una acción concreta.}\\[8pt]

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{doc-blocker} — \texttt{PreToolUse(Write)}\\[3pt]
      \small Bloquea \texttt{.md/.txt} espurios (Claude tiende a escupir READMEs no pedidos). Permite nombres canónicos (\texttt{README.md}, \texttt{CLAUDE.md}) y rutas estructuradas (\texttt{.claude/}, \texttt{docs/}).
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{git-push-guard} — \texttt{PreToolUse(Bash)}\\[3pt]
      \small Intercepta el tool \texttt{Bash} y bloquea un \texttt{git push --force} (y variantes) antes de que se ejecute. Exit 2 $\rightarrow$ Claude ve el motivo y reintenta distinto.
    \end{column}
  \end{columns}
  \vspace{10pt}
  \footnotesize\textcolor{muted}{\textbf{Heurística:} un hook por cada acción que quiero enforzar de forma determinística — algo que ya me pasó y no quiero que se repita. Nada especulativo.}
\end{frame}

% ─── S8 · Skills ─────────────────────────────────────────────────────────────
\begin{frame}{Skills\rf{6}}

  {\small \quees un workflow empaquetado (instrucciones + tools + scripts) que Claude carga \textbf{on-demand}. Vive en \texttt{.claude/skills/<name>/SKILL.md}.}\\[3pt]
  {\small \paraque reutilizar procesos con \texttt{/<name> [args]}, o que Claude los auto-invoque según su \texttt{description}.}\\[6pt]

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Frontmatter clave}\\[3pt]
      \small
      \begin{tabular}{ll}
        \texttt{description}      & cuándo auto-invoca \\
        \texttt{argument-hint}    & hint de autocomplete \\
        \texttt{allowed-tools}    & tools pre-aprobadas \\
        \texttt{model} / \texttt{effort} & overrides \\
        \texttt{context: fork}    & corre en subagente \\
      \end{tabular}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Inyección dinámica en el body}\\[3pt]
      \small \texttt{\$ARGUMENTS} interpola los args. \texttt{!\`{}cmd\`{}} corre el comando e inyecta el output \textit{antes} de que Claude vea la skill.
      \vspace{6pt}
      \footnotesize \texttt{description} es un \textbf{trigger}, no un resumen — Claude lo usa para decidir cuándo auto-invocar.
    \end{column}
  \end{columns}
\end{frame}

% ─── S9 · Plugins ────────────────────────────────────────────────────────────
\begin{frame}{Plugins\rf{7}}

  {\small \quees un paquete que agrupa varias superficies (skills + agents + hooks + MCP) en una unidad instalable.}\\[3pt]
  {\small \paraque compartir workflow con el equipo sin copiar-pegar configs.}\\[6pt]

  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Qué puede contener}\\[3pt]
      \small
      \begin{itemize}
        \item \texttt{.claude-plugin/plugin.json} — nombre, versión
        \item \texttt{skills/}\dd \texttt{agents/}\dd \texttt{hooks/hooks.json}
        \item \texttt{.mcp.json}\dd \texttt{lsp/}\dd \texttt{output-styles/}
      \end{itemize}
      \vspace{4pt}
      \textbf{Instalar}\\[2pt]
      \small \texttt{claude plugin install <name>@<marketplace>}
    \end{column}
    \begin{column}{0.44\textwidth}
      \small Puede ser tan simple como una carpeta con \texttt{SKILL.md} + \texttt{plugin.json}.
    \end{column}
  \end{columns}
\end{frame}

% ─── S10 · MCP Servers ───────────────────────────────────────────────────────
\begin{frame}{MCP Servers\rf{8}}

  {\small \quees Model Context Protocol: extiende a Claude con tools de sistemas externos.}\\[3pt]
  {\small \paraque llegar más allá del repo — Jira, Gmail, Calendar, Drive…}\\[6pt]

  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \textbf{3 transports}\\[3pt]
      \small
      \begin{tabular}{ll}
        \texttt{stdio} & proceso local, stdin/stdout \\
        \texttt{sse}   & Server-Sent Events (URL) \\
        \texttt{http}  & HTTP POST (URL) \\
      \end{tabular}
      \vspace{5pt}
      \textbf{Config por scope}\\[2pt]
      \small
      \begin{itemize}
        \item \texttt{\textasciitilde/.claude.json} — user
        \item \texttt{.mcp.json} — proyecto, commiteado
        \item managed — org-wide
      \end{itemize}
    \end{column}
    \begin{column}{0.50\textwidth}
      \textbf{Ejemplo}\\[3pt]
      \small
      \begin{itemize}
        \item \textbf{Jira} (remote, OAuth) — issues, PRs, transiciones
        \item \textbf{gogcli} (local stdio) — Gmail, Calendar, Drive, Docs, Sheets
      \end{itemize}
      \vspace{6pt}
      \footnotesize\textcolor{muted}{Secrets nunca en disco: OAuth en runtime o \texttt{env} del proceso.}
    \end{column}
  \end{columns}
\end{frame}

% ─── S11 · Status Line + Keybindings ─────────────────────────────────────────
\begin{frame}{Status Line y Keybindings\rf{9}}

  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Status Line}\\[3pt]
      {\small \quees un script \textbf{tuyo} que Claude corre en cada render, pasándole el estado de la sesión por stdin (JSON).}\\[2pt]
      {\small \paraque ver de un vistazo el estado.}\\[4pt]
      \small
      \begin{tabular}{ll}
        \texttt{model.display\_name} & modelo activo \\
        \texttt{context\_window.*}   & tokens / \% \\
        \texttt{session.cost\_usd}   & costo acumulado \\
        \texttt{vim.mode}            & insert / normal \\
      \end{tabular}
      \vspace{3pt}
      \footnotesize\alert{El \% de contexto es el campo clave} — Claude empieza a ignorar instrucciones cerca del 40\%.
    \end{column}
    \begin{column}{0.44\textwidth}
      \textbf{Keybindings}\\[3pt]
      {\small \quees atajos personalizados en \texttt{\textasciitilde/.claude/keybindings.json} (hot-reload).}\\[2pt]
      {\small \paraque adaptar la navegación.}\\[4pt]
      \small
      \begin{itemize}
        \item $\sim$19 contextos (\texttt{Chat}, \texttt{Autocomplete}…)
        \item $\sim$80 acciones en namespaces (\texttt{chat:*}, \texttt{app:*}…)
        \item Chords: \texttt{ctrl+k ctrl+s}
        \item \texttt{null} desactiva un binding
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ─── S12 · Anti-slop ─────────────────────────────────────────────────────────
\begin{frame}{Anti-slop — Ponytail / Caveman / RTK\rf{10}}

  {\small \quees tres herramientas que ajustan el comportamiento por un ángulo distinto cada una.}\\[6pt]

  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \textbf{Ponytail} \hfill \texttt{/ponytail}\\[3pt]
      \small \textit{Qué construye Claude}\\[2pt]
      Modo YAGNI:
      \begin{itemize}
        \item Stdlib antes que librería
        \item La solución más corta que funciona
        \item Sin abstracciones no pedidas
      \end{itemize}
    \end{column}
    \begin{column}{0.32\textwidth}
      \textbf{Caveman} \hfill \texttt{/caveman}\\[3pt]
      \small \textit{Cómo habla Claude}\\[2pt]
      Prosa terse:
      \begin{itemize}
        \item Respuestas de 1-2 líneas
        \item Sin resúmenes al final
        \item Sin explicar lo obvio del diff
      \end{itemize}
    \end{column}
    \begin{column}{0.32\textwidth}
      \textbf{RTK} \hfill \small\textit{proxy CLI}\\[3pt]
      \small \textit{Cuántos tokens gasta bash}\\[2pt]
      Comprime output de comandos:
      \begin{itemize}
        \item Filtra, agrupa, trunca, deduplica
        \item Hasta 90\% menos tokens
        \item 100+ comandos (git, aws…)
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{6pt}
  \footnotesize\textcolor{muted}{\paraque menos código inútil, menos prosa inútil, menos tokens inútiles.}
\end{frame}

% ─── S13 · Static tools — Python ─────────────────────────────────────────────
\begin{frame}{Static tools — Python\rf{11}}

  {\small \quees linters y formatters que corren en \textbf{pre-commit}, con o sin Claude.}\\[3pt]
  {\small \paraque que el código cumpla el estándar antes del commit. Claude lee el output si falla, corrige y reintenta solo.}\\[8pt]

  \textbf{Stack} — framework \texttt{pre-commit}\\[4pt]
  \small
  \begin{tabular}{lll}
    \toprule
    \textbf{Tool} & \textbf{Qué hace} & \textbf{Dónde} \\
    \midrule
    \textbf{Ruff}  & lint + format en un binario (reemplaza flake8 + black + isort) & pre-commit \\
    \textbf{mypy}  & chequeo de tipos & pre-commit \\
    \bottomrule
  \end{tabular}
  \vspace{8pt}
  \footnotesize\textcolor{muted}{Regla general: lo rápido y determinístico bloquea el commit; lo lento informa en el PR.}
\end{frame}

% ─── S14 · Static tools — Java ───────────────────────────────────────────────
\begin{frame}{Static tools — Java\rf{11}}

  {\small \quees el mismo enforcement por pre-commit + CI, adaptado a Java.}\\[3pt]
  {\small \paraque formato y estilo bloquean el commit; el análisis pesado informa en el PR.}\\[8pt]

  \small
  \begin{tabular}{lll}
    \toprule
    \textbf{Tool} & \textbf{Qué hace} & \textbf{Dónde} \\
    \midrule
    \textbf{Spotless}   & formato (google-java-format) + imports & pre-commit (rápido) \\
    \textbf{Checkstyle} & naming, longitud, imports & pre-commit, bloqueante \\
    \textbf{SpotBugs}   & null deref, resource leaks, races & CI (lento) \\
    \textbf{PMD}        & complejidad, código muerto & CI \\
    \bottomrule
  \end{tabular}
  \vspace{8pt}
  \footnotesize La skill \texttt{/java-quarkus} sabe qué errores de SpotBugs son bloqueantes vs warnings.
\end{frame}

% ─── S15 · Static tools — TypeScript ─────────────────────────────────────────
\begin{frame}{Static tools — TypeScript\rf{11}}

  {\small \quees el mismo enforcement, vía \textbf{Husky + lint-staged}.}\\[3pt]
  {\small \paraque bloquear el commit con lo rápido; dejar lo lento/ruidoso para CI.}\\[8pt]

  \small
  \begin{tabular}{lll}
    \toprule
    \textbf{Tool} & \textbf{Qué hace} & \textbf{Dónde} \\
    \midrule
    \textbf{prettier}     & formato sobre staged & pre-commit (lint-staged) \\
    \textbf{tsc --noEmit} & typecheck & pre-commit \\
    \textbf{tests}        & suite en el commit & pre-commit \\
    \textbf{ESLint}       & lint completo & CI (lento/ruidoso) \\
    \bottomrule
  \end{tabular}
\end{frame}
